name: Build PSP app
on:
  push:
  pull_request:
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    container: pspdev/pspdev:latest
    steps:
      - uses: actions/checkout@v4
      - name: Build EBOOT.PBP
        run: |
          export PSPDEV=/usr/local/pspdev
          export PATH=$PATH:$PSPDEV/bin
          make -j$(nproc)
          mkdir -p dist/PKStorage
          cp EBOOT.PBP dist/PKStorage/
          # Future: copy sprite atlas here when ready
          # cp assets/sprites.bin dist/PKStorage/
      - uses: actions/upload-artifact@v4
        with:
          name: PKStorage-psp
          path: dist
          if-no-files-found: error
